\frame{
    \frametitle{Ordering Descriptors}

    \begin{columns}
        \column{.45\textwidth}
        \onslide<1->{
        \begin{block}{$D_1 \leq D_2$}
            For every $K$,
            $$ \mu(D_2,K) \leq \mu(D_1,K).$$
        \end{block}
        }

        \onslide<2->{
        \begin{block}{$D_1 < D_2$}
            \begin{enumerate}
                \item $D_1 \leq D_2$
                \item there exists a complex $K$ where $\mu(D_2,K) < \mu(D_1,K)$.
            \end{enumerate}
        \end{block}
        }

        \column{.45\textwidth}
        \onslide<1->{

\frame{
    \frametitle{(Verbose) Euler Characteristic Functions}

    \begin{columns}
        \column{.45\textwidth}

        \begin{block}{}
            Let $h \colon K \to \R$ be a filter.
        \end{block}

        \onslide<2->{
        \begin{block}{Euler Char.\ Function (ECF)}
            A function $\chi \colon \R \to \N$ defined by
            $$\chi(t) = \sum_{\sigma \in h^{-1}(-\infty,t]} (-1)^{\dim(\sigma)}.$$
        \end{block}
        }

        \onslide<3->{
        \begin{block}{Verbose ECF}
            A function $\hat{\chi} \colon \R \to \N^2$ defined by
            $$\hat{\chi}(t) = ( \# (+1)s,  \# (-1)s ). $$
        \end{block}
        }

\frame{
    \frametitle{Verbose Is No Worse}

    \begin{block}{}
        \centering
        Descriptor $\leq$ Verbose Descriptor
    \end{block}

    \pause
    \begin{block}{Why?}
        Ignore (forget) some info, and we have the usual descriptor.
    \end{block}

    \pause
    \begin{block}{But Wait!}
        The inequality is actually strict!\\
        \begin{center}
            Descriptor $<$ Verbose Descriptor
        \end{center}
    \end{block}

    \pause
    \begin{block}{Why?}
        Again, a single edge.
    \end{block}
}
